create or replace function public.handle_new_user()
returns trigger as $$
begin
  insert into public.profiles (id, nome, telefone, role)
  values (
    new.id,
    coalesce(new.raw_user_meta_data->>'nome', 'Usuário'),
    new.raw_user_meta_data->>'telefone',
    'cliente'
  );
  return new;
end;

create trigger on_auth_user_created
  after insert on auth.users

-- ─── FUNÇÃO: buscar email pelo telefone (para login com tel) ──
create or replace function public.get_email_by_phone(p_telefone text)
returns text as $$
  select u.email
  from public.profiles pr
  join auth.users u on u.id = pr.id
  where pr.telefone = p_telefone
  limit 1;
$$ language sql security definer;

Não! No Vercel você não precisa colocar a anon key como variável de ambiente nesse caso.

O motivo é simples: a anon key do Supabase é pública por design — ela já fica exposta no código do frontend que vai pro navegador do usuário de qualquer forma. Não tem segredo nenhum nela; o que protege seus dados são as políticas de RLS que você já configurou no banco.

Então é só fazer o deploy normalmente no Vercel que vai funcionar. O esm.sh busca o pacote do Supabase direto na CDN, sem precisar de npm install nem de nada extra.

Se em algum momento você tiver uma service role key (que é secreta e tem acesso total ao banco, bypassando o RLS) — aí sim, essa você nunca coloca no frontend e usaria como variável de ambiente numa função serverless. Mas por enquanto não é o caso.

sqlcreate table public.agendamentos (
  id           uuid primary key default gen_random_uuid(),
  cliente_id   uuid not null references auth.users(id) on delete cascade,
  filial_id    int  not null references public.filiais(id),
  barbeiro     text not null,
  servico      text not null,
  preco        text not null,
  horario      timestamptz not null,
  observacoes  text,
  status       text not null default 'confirmado'
                check (status in ('confirmado','pendente','cancelado')),
  created_at   timestamptz not null default now()
);

alter table public.agendamentos enable row level security;

-- Cliente vê e cria apenas os próprios agendamentos
create policy "cliente: leitura própria"
  on public.agendamentos for select
  using (auth.uid() = cliente_id);

create policy "cliente: inserir"
  on public.agendamentos for insert
  with check (auth.uid() = cliente_id);

-- Leitura de horários ocupados (necessário para o select de disponibilidade)
create policy "público: ver horários ocupados"
  on public.agendamentos for select
